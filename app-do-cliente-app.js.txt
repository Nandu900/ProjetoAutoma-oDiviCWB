function mulberry32(a){return function(){a|=0;a=a+0x6D2B79F5|0;let t=Math.imul(a^a>>>15,1|a);t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296}}

// ocupação de exemplo para os próximos 21 dias: é o que sobra que vira horário livre
const OCUPADO = [];
(function(){
  for(let k=0;k<21;k++){
    const d = addDias(HOJE,k), dia = iso(d);
    PROS.forEach(p=>{
      const j = jornada(p,d);
      if(!j) return;
      OCUPADO.push({data:dia, proId:p.id, ini: p.id==="p2"?13*60:12*60, dur:60, motivo:"Almoço"});
      const carga = k===0 ? .58 : k<=2 ? .46 : k<=6 ? .34 : .2;   // dias próximos mais cheios
      let cursor = j[0];
      while(cursor < j[1]-30){
        const dur = [30,45,60,60,90,120][ri(0,5)];
        if(rnd() < carga && cursor+dur <= j[1]){
          if(!(cursor < (p.id==="p2"?14*60:13*60) && cursor+dur > (p.id==="p2"?13*60:12*60)))
            OCUPADO.push({data:dia, proId:p.id, ini:cursor, dur:dur, motivo:"ocupado"});
          cursor += dur;
        } else cursor += 30;
      }
    });
  }
})();

function conflita(dia, proId, ini, dur){
  return OCUPADO.some(o => o.data===dia && o.proId===proId && ini < o.ini+o.dur && ini+dur > o.ini)
      || MEUS.some(m => m.data===dia && m.proId===proId && m.status==="marcado" &&
                        ini < m.ini+m.dur && ini+dur > m.ini);
}

// horários livres de um serviço num dia; proId null = qualquer profissional
function livres(servicoId, proId, d){
  const s = srv(servicoId), dia = iso(d), fora = [];
  const lista = (proId ? [pro(proId)] : s.pros.map(pro));
  const mapa = {};
  lista.forEach(p=>{
    const j = jornada(p,d);
    if(!j) return;
    for(let m=j[0]; m+s.dur<=j[1]; m+=30){
      if(iso(d)===iso(HOJE) && m < AGORA_MIN+60) continue;   // nada em cima da hora
      if(conflita(dia, p.id, m, s.dur)) continue;
      if(!mapa[m]) mapa[m] = p.id;
    }
  });
  return Object.keys(mapa).map(Number).sort((a,b)=>a-b).map(m=>({ini:m, proId:mapa[m]}));
}
function temVaga(servicoId, proId, d){ return livres(servicoId, proId, d).length > 0; }

function pintaApp(){
  let tela;
  if(confirmado) tela = telaConfirmado();
  else if(aba==="inicio") tela = telaInicio();
  else if(aba==="agendar") tela = telaAgendar();
  else if(aba==="meus") tela = telaMeus();
  else tela = telaFidelidade();

  fone.innerHTML = '<div class="tela" id="tela">'+tela+'</div>'+
    '<nav class="rodape" aria-label="Seções">'+
      ABAS.map(([k,rot,d])=>'<button data-aba="'+k+'" aria-current="'+(!confirmado && aba===k)+'">'+
        '<svg class="pt" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.7" stroke-linecap="round" stroke-linejoin="round">'+d+'</svg>'+rot+'</button>').join("")+
    '</nav>';

  const f = document.getElementById("form-confirmar");
  if(f) f.addEventListener("submit", confirmarPagina);
  legenda.textContent = "O mesmo link que o salão põe na bio do Instagram e no Google. Nenhum cadastro, nenhum aplicativo para instalar.";
}

function confirmarPagina(ev){
  ev.preventDefault();
  const aviso = document.getElementById("f-aviso");
  const nome = (document.getElementById("f-nome").value||"").trim();
  const tel = (document.getElementById("f-tel").value||"").replace(/\D/g,"");
  if(nome.length < 3){ aviso.hidden=false; aviso.textContent="Escreva seu nome completo."; return; }
  if(tel.length < 10){ aviso.hidden=false; aviso.textContent="Confira o número do WhatsApp, com DDD."; return; }
  const d = new Date(passo.data+"T12:00:00");
  const proEscolhido = passo.proId || (livres(passo.servicoId, null, d).find(v=>v.ini===passo.ini)||{}).proId;
  if(!proEscolhido){ aviso.hidden=false; aviso.textContent="Esse horário acabou de ser preenchido. Escolha outro."; return; }
  EU.nome = nome;
  confirmado = marcar(passo.servicoId, proEscolhido, passo.data, passo.ini, "Página");
  passo = null;
  pintaApp();
  document.getElementById("tela").scrollTop = 0;
}

      (digitando?'<div class="digitando"><i></i><i></i><i></i></div>':'')+'</div>'+

function botDiz(textos, proximas, espera){
  digitando = true; rapidas = []; pintaZap();
  setTimeout(()=>{
    digitando = false;
    (Array.isArray(textos)?textos:[textos]).forEach(t=>diz("bot", t));
    rapidas = proximas || [];
    pintaZap();
  }, espera || 620);
}

function iniciaZap(){
  conversa = []; rapidas = []; estadoZap = {};
  diz("bot", "Oi! Aqui é o "+SALAO.nome+" 💈\nPosso te ajudar com o quê?");
  rapidas = [
    {rot:"Quero marcar um horário", acao:"marcar"},
    {rot:"Ver meus horários", acao:"ver"},
    {rot:"Falar com atendente", acao:"humano", fraca:true}
  ];
  pintaZap();
}

function zapEscolhe(i){
  const r = rapidas[i];
  if(!r) return;
  diz("eu", r.rot);
  rapidas = [];
  pintaZap();
  setTimeout(()=>trataZap(r), 260);
}

function trataZap(r){
  const st = estadoZap;

  if(r.acao==="humano"){
    botDiz("Combinado, já chamei alguém do salão. Enquanto isso, se quiser adiantar, é só me dizer o serviço.",
      [{rot:"Quero marcar um horário", acao:"marcar"}, {rot:"Recomeçar", acao:"reset", fraca:true}]);
    return;
  }

  if(r.acao==="reset"){ iniciaZap(); return; }

  if(r.acao==="ver"){
    const p = proximos();
    if(!p.length){
      botDiz("Não achei horário marcado no seu número. Quer marcar um agora?",
        [{rot:"Quero marcar um horário", acao:"marcar"}]);
    } else {
      const linhas = p.map(m=>{
        const s = srv(m.servicoId), pr = pro(m.proId), d = new Date(m.data+"T12:00:00");
        return "• "+s.nome+" — "+dataLonga(d)+" às "+hm(m.ini)+", com "+pr.nome.split(" ")[0];
      }).join("\n");
      botDiz("Você tem:\n"+linhas+"\n\nQuer fazer mais alguma coisa?",
        [{rot:"Marcar outro horário", acao:"marcar"}, {rot:"Só isso, obrigada", acao:"tchau", fraca:true}]);
    }
    return;
  }

  if(r.acao==="tchau"){
    botDiz("Então tá! Te espero por aqui 💚",
      [{rot:"Recomeçar a conversa", acao:"reset", fraca:true}]);
    return;
  }

  if(r.acao==="marcar"){
    st.servicoId = null; st.proId = null; st.data = null; st.ini = null;
    const populares = ["s7","s8","s1","s2","s10","s12"].map(srv);
    botDiz("Boa! Qual serviço você quer?",
      populares.map(s=>({rot:s.nome+" · "+BRL(s.preco), acao:"srv", id:s.id}))
        .concat([{rot:"Ver todos os serviços", acao:"todos", fraca:true}]));
    return;
  }

  if(r.acao==="todos"){
    botDiz("Nossa lista completa:", SERVICOS.map(s=>({rot:s.nome+" · "+BRL(s.preco), acao:"srv", id:s.id})));
    return;
  }

  if(r.acao==="srv"){
    st.servicoId = r.id;
    const s = srv(r.id);
    if(s.pros.length===1){
      st.proId = s.pros[0];
      const p = pro(st.proId);
      botDiz(s.nome+" leva "+s.dur+" minutos e sai por "+BRL(s.preco)+". Quem atende é "+p.nome.split(" ")[0]+".\nPara quando?",
        diasRapidos(st));
    } else {
      botDiz("Com quem você prefere?",
        [{rot:"Tanto faz", acao:"pro", id:null}].concat(s.pros.map(id=>({rot:pro(id).nome.split(" ")[0], acao:"pro", id}))));
    }
    return;
  }

  if(r.acao==="pro"){
    st.proId = r.id;
    botDiz("Fechado. Para quando?", diasRapidos(st));
    return;
  }

  if(r.acao==="dia"){
    st.data = r.id;
    const d = new Date(r.id+"T12:00:00");
    const vagas = livres(st.servicoId, st.proId, d);
    if(!vagas.length){
      botDiz("Esse dia lotou. Tenho estes outros:", diasRapidos(st));
      return;
    }
    const amostra = vagas.filter((v,i)=> i%Math.max(1,Math.floor(vagas.length/6))===0).slice(0,6);
    botDiz("Tenho estes horários "+dataLonga(d)+":",
      amostra.map(v=>({rot:hm(v.ini), acao:"hora", id:v.ini}))
        .concat([{rot:"Outro dia", acao:"outrodia", fraca:true}]));
    return;
  }

  if(r.acao==="outrodia"){ botDiz("Sem problema. Qual dia?", diasRapidos(st)); return; }

  if(r.acao==="hora"){
    st.ini = r.id;
    const s = srv(st.servicoId), d = new Date(st.data+"T12:00:00");
    const escolhido = st.proId || (livres(st.servicoId, null, d).find(v=>v.ini===st.ini)||{}).proId;
    st.proEscolhido = escolhido;
    botDiz("Deixa eu confirmar:\n\n"+s.nome+"\n"+dataLonga(d)+" às "+hm(st.ini)+
           "\ncom "+pro(escolhido).nome+"\n"+BRL(s.preco)+"\n\nPosso fechar?",
      [{rot:"Pode fechar", acao:"fechar"}, {rot:"Quero trocar o horário", acao:"outrodia", fraca:true}]);
    return;
  }

  if(r.acao==="fechar"){
    const m = marcar(st.servicoId, st.proEscolhido, st.data, st.ini, "WhatsApp");
    const d = new Date(m.data+"T12:00:00");
    const n = feitos().length % SALAO.metaFidelidade;
    botDiz(["Prontinho, "+EU.nome.split(" ")[0]+"! Seu horário está marcado ✅",
            "Te lembro um dia antes por aqui. Se precisar desmarcar, é só me chamar.\n\n"+
            SALAO.endereco+"\n\nAh: você está com "+n+" de "+SALAO.metaFidelidade+" carimbos no cartão fidelidade."],
      [{rot:"Ver meus horários", acao:"ver"}, {rot:"Recomeçar", acao:"reset", fraca:true}], 800);
    return;
  }
}

function diasRapidos(st){
  const out = [];
  for(let k=0;k<14 && out.length<5;k++){
    const d = addDias(HOJE,k);
    if(!SALAO.abre[d.getDay()]) continue;
    if(!temVaga(st.servicoId, st.proId, d)) continue;
    out.push({rot: k===0?"Hoje":k===1?"Amanhã":WD[d.getDay()]+" "+d.getDate()+"/"+pad(d.getMonth()+1),
              acao:"dia", id:iso(d)});
  }
  if(!out.length) out.push({rot:"Falar com atendente", acao:"humano", fraca:true});
  return out;
}

/* ============================ eventos ============================ */
document.addEventListener("click", e=>{
  const t = e.target.closest("[data-canal],[data-aba],[data-srv],[data-pro],[data-dia],[data-hora],[data-voltar],[data-ir],[data-cancelar],[data-remarcar],[data-repetir],[data-zap]");
  if(!t) return;

  if(t.dataset.canal){
    canal = t.dataset.canal;
    document.querySelectorAll("[data-canal]").forEach(b=>b.setAttribute("aria-pressed", String(b.dataset.canal===canal)));
    if(canal==="zap") iniciaZap(); else pintaApp();
    return;
  }
  if(t.dataset.zap !== undefined){ zapEscolhe(parseInt(t.dataset.zap,10)); return; }

  if(t.dataset.aba){
    confirmado = null;
    aba = t.dataset.aba;
    if(aba==="agendar" && !passo) passo = {n:1, servicoId:null, proId:null, data:null, ini:null};
    pintaApp(); return;
  }
  if(t.dataset.ir){
    confirmado = null; aba = t.dataset.ir;
    if(aba==="agendar") passo = {n:1, servicoId:null, proId:null, data:null, ini:null};
    pintaApp(); return;
  }
  if(t.dataset.srv){
    passo = {n:2, servicoId:t.dataset.srv, proId:null, data:null, ini:null};
    const s = srv(t.dataset.srv);
    if(s.pros.length===1){ passo.proId = s.pros[0]; passo.n = 3; }
    aba = "agendar"; confirmado = null; pintaApp(); return;
  }
  if(t.dataset.pro){
    passo.proId = t.dataset.pro==="qualquer" ? null : t.dataset.pro;
    passo.n = 3; passo.data = null; passo.ini = null; pintaApp(); return;
  }
  if(t.dataset.dia){ passo.data = t.dataset.dia; passo.ini = null; passo.n = 4; pintaApp(); return; }
  if(t.dataset.hora){ passo.ini = parseInt(t.dataset.hora,10); passo.n = 5; pintaApp(); return; }
  if(t.dataset.voltar){
    if(passo.n<=1){ aba = "inicio"; passo = null; }
    else { passo.n -= 1; if(passo.n===2 && srv(passo.servicoId).pros.length===1) passo.n = 1; }
    pintaApp(); return;
  }
  if(t.dataset.cancelar){
    const m = MEUS.find(x=>x.id===t.dataset.cancelar);
    if(m) m.status = "cancelado";
    pintaApp(); return;
  }
  if(t.dataset.remarcar){
    const m = MEUS.find(x=>x.id===t.dataset.remarcar);
    if(m){ m.status = "cancelado";
      passo = {n:3, servicoId:m.servicoId, proId:m.proId, data:null, ini:null};
      aba = "agendar"; confirmado = null; }
    pintaApp(); return;
  }
  if(t.dataset.repetir){
    const s = srv(t.dataset.repetir);
    passo = {n: s.pros.length===1?3:2, servicoId:s.id, proId: s.pros.length===1?s.pros[0]:null, data:null, ini:null};
    aba = "agendar"; confirmado = null; pintaApp(); return;
  }
});
